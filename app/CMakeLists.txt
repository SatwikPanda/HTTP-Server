add_executable(http_app main.cpp routes.cpp)
target_include_directories(http_app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(http_app PRIVATE application)
target_compile_features(http_app PRIVATE cxx_std_20)
set_target_properties(http_app PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(http_app PRIVATE /W4)
else()
    target_compile_options(http_app PRIVATE -Wall -Wextra -Wpedantic)
endif()
